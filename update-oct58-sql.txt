--   الأعضاء الأساسيون لمحادثة المشروع (تلقائيًا): منشئ الميثاق، مدير المشروع، المالك.
--   كل من يصبح طرفًا لاحقًا (مكلّف بمهمة، موقّع، مستشار، صاحب صلاحية) يُرفع طلب انضمامه لمدير المشروع ومكتب التطوير.
--   الأعضاء الحاليون في المحادثات القائمة يُحفظون كما هم (لا يفقد أحد وصوله).

-- ===== (١) تفاعلات الإيموجي =====
create table if not exists public.chat_reactions(
  msg_id bigint not null references public.chat_messages(id) on delete cascade,
  user_id uuid not null default auth.uid(),
  emoji text not null check (char_length(emoji) between 1 and 16),
  at timestamptz not null default now(),
  primary key (msg_id, user_id));
alter table public.chat_reactions enable row level security;
drop policy if exists "crx sel" on public.chat_reactions;
create policy "crx sel" on public.chat_reactions for select to authenticated using (exists(select 1 from public.chat_messages m where m.id = msg_id));
grant select on public.chat_reactions to authenticated;

create or replace function public.chat_react(p_msg bigint, p_emoji text) returns void
language plpgsql security definer set search_path=public as $fn$
begin
  if not exists(select 1 from chat_messages m where m.id = p_msg and not m.deleted
                and (m.from_user = auth.uid() or m.to_user = auth.uid() or (m.channel_id is not null and chat_can_access(m.channel_id)))) then
    raise exception 'not allowed'; end if;
  if coalesce(trim(p_emoji),'') = '' then delete from chat_reactions where msg_id = p_msg and user_id = auth.uid(); return; end if;
  insert into chat_reactions(msg_id,user_id,emoji) values (p_msg, auth.uid(), left(trim(p_emoji),16))
  on conflict (msg_id,user_id) do update set emoji = excluded.emoji, at = now();
end $fn$;
grant execute on function public.chat_react(bigint,text) to authenticated;

-- ===== (٢) العضوية الديناميكية =====
-- حفظ الأعضاء الحاليين في المحادثات القائمة قبل تغيير القاعدة
insert into public.chat_members(channel_id, user_id, added_by)
select c.id, u, null from public.chat_channels c, public.chat_project_members(c.pcode) u
where c.kind = 'project' and c.pcode is not null
on conflict do nothing;

-- كل من له علاقة بالمشروع ولماذا
create or replace function public.chat_involved(p_pcode text) returns table(user_id uuid, why text)
language sql stable security definer set search_path=public as $$
  select distinct on (u) u, w from (
    select pl.created_by u, 'أعدّ خطة المشروع' w from plans pl where pl.data->>'pcode'=p_pcode and not pl.archived
    union all select s.user_id, 'موقّع على الميثاق' from charter_signatories s join charters c on c.id=s.cr_id where c.code=p_pcode
    union all select s.user_id, 'موقّع على الخطة' from plan_signatories s join plans pl on pl.id=s.cr_id where pl.data->>'pcode'=p_pcode
    union all select case when t->>'w' ~ '^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$' then (t->>'w')::uuid end, 'مكلّف بمهمة: '||left(coalesce(t->>'n',''),60)
              from plans pl, jsonb_array_elements(coalesce(pl.data->'tasks','[]'::jsonb)) t where pl.data->>'pcode'=p_pcode and not pl.archived
    union all select g.user_id, 'مُنح صلاحية على المشروع' from project_grants g where g.pcode=p_pcode
    union all select unnest(k.invitees), 'مشارك في مراجعة الخطة' from plan_consult k join plans pl on pl.id=k.plan_id where pl.data->>'pcode'=p_pcode
  ) z where u is not null order by u
$$;
revoke all on function public.chat_involved(text) from public;

-- الأعضاء الأساسيون فقط (تلقائيًا): منشئ الميثاق، مدير المشروع، المالك
create or replace function public.chat_project_members(p_pcode text) returns setof uuid
language sql stable security definer set search_path=public as $$
  select distinct u from (
    select c.created_by u from charters c where c.code=p_pcode
    union all select p.id from profiles p join charters c on c.code=p_pcode and (p.full_name=c.data->>'pmname' or p.full_name=c.data->>'owner')
    union all select p.id from profiles p join plans pl on pl.data->>'pcode'=p_pcode and not pl.archived and p.full_name=pl.data->>'pmname'
  ) z where u is not null and not exists(select 1 from chat_excl x where x.pcode = p_pcode and x.user_id = z.u)
$$;
revoke all on function public.chat_project_members(text) from public;

-- هل أنا مدير المشروع أو مالكه؟
create or replace function public.chat_is_pm(p_pcode text) returns boolean
language sql stable security definer set search_path=public as $$
  select exists(select 1 from profiles me where me.id = auth.uid() and (
      exists(select 1 from charters c where c.code = p_pcode and (c.data->>'pmname' = me.full_name or c.data->>'owner' = me.full_name))
   or exists(select 1 from plans pl where pl.data->>'pcode' = p_pcode and not pl.archived and pl.data->>'pmname' = me.full_name)))
$$;
grant execute on function public.chat_is_pm(text) to authenticated;

-- قناة المشروع (تُنشأ عند الحاجة)
create or replace function public._chat_ch(p_pcode text) returns bigint
language plpgsql security definer set search_path=public as $fn$
declare i bigint; nm text;
begin
  select id into i from chat_channels where kind='project' and pcode=p_pcode;
  if i is null then
    nm := coalesce((select data->>'pname' from plans where data->>'pcode'=p_pcode order by id desc limit 1),(select data->>'pname' from charters where code=p_pcode order by id desc limit 1),p_pcode);
    insert into chat_channels(kind,pcode,name) values('project',p_pcode,nm) on conflict do nothing returning id into i;
    if i is null then select id into i from chat_channels where kind='project' and pcode=p_pcode; end if;
  end if;
  return i;
end $fn$;
revoke all on function public._chat_ch(text) from public;

-- طلبات الانضمام
create table if not exists public.chat_join_requests(
  id bigserial primary key,
  pcode text not null, user_id uuid not null,
  why text, source text not null default 'auto' check (source in ('auto','request')),
  requested_by uuid, status text not null default 'pending' check (status in ('pending','approved','rejected')),
  decided_by uuid, decided_at timestamptz, note text,
  created_at timestamptz not null default now());
create unique index if not exists chat_jr_pend on public.chat_join_requests(pcode,user_id) where status='pending';
alter table public.chat_join_requests enable row level security;
drop policy if exists "cjr sel" on public.chat_join_requests;
create policy "cjr sel" on public.chat_join_requests for select to authenticated
  using (exists(select 1 from profiles where id=auth.uid() and role='admin') or chat_is_pm(pcode) or user_id=auth.uid() or requested_by=auth.uid());
grant select on public.chat_join_requests to authenticated;

-- تنبيه مدير المشروع ومكتب التطوير
create or replace function public._chat_notify_mgrs(p_pcode text, p_msg text) returns void
language plpgsql security definer set search_path=public as $fn$
declare me text;
begin
  select full_name into me from profiles where id=auth.uid();
  insert into user_nudges(to_user,from_user,from_name,msg)
  select distinct p.id, coalesce(auth.uid(),(select id from profiles where role='admin' limit 1)), coalesce(me,'المنصة'), left(p_msg,480) from profiles p
  where p.id is distinct from auth.uid() and (p.role='admin'
     or exists(select 1 from charters c where c.code=p_pcode and (c.data->>'pmname'=p.full_name or c.data->>'owner'=p.full_name))
     or exists(select 1 from plans pl where pl.data->>'pcode'=p_pcode and not pl.archived and pl.data->>'pmname'=p.full_name));
end $fn$;
revoke all on function public._chat_notify_mgrs(text,text) from public;

-- رصد الأطراف الجدد في مشروع وإنشاء طلبات انضمامهم (مرة واحدة لكل شخص)
create or replace function public.chat_sync(p_pcode text) returns int
language plpgsql security definer set search_path=public as $fn$
declare n int := 0; ch bigint; r record; pn text;
begin
  if coalesce(p_pcode,'') = '' then return 0; end if;
  select id into ch from chat_channels where kind='project' and pcode=p_pcode;
  for r in select i.user_id, i.why from chat_involved(p_pcode) i
           where i.user_id not in (select public.chat_project_members(p_pcode))
             and not exists(select 1 from chat_members m where m.channel_id=ch and m.user_id=i.user_id)
             and not exists(select 1 from chat_excl x where x.pcode=p_pcode and x.user_id=i.user_id)
             and not exists(select 1 from chat_join_requests q where q.pcode=p_pcode and q.user_id=i.user_id)
             and exists(select 1 from profiles where id=i.user_id) loop
    insert into chat_join_requests(pcode,user_id,why,source) values (p_pcode,r.user_id,r.why,'auto') on conflict do nothing;
    n := n + 1;
  end loop;
  if n > 0 then
    pn := coalesce((select data->>'pname' from charters where code=p_pcode order by id desc limit 1), p_pcode);
    perform _chat_notify_mgrs(p_pcode, n||' أشخاص صاروا طرفًا في مشروع «'||pn||'». هل يُضافون إلى محادثة المشروع؟ راجع طلبات الانضمام في التواصل.');
  end if;
  return n;
end $fn$;
grant execute on function public.chat_sync(text) to authenticated;

create or replace function public.chat_sync_all() returns int
language plpgsql security definer set search_path=public as $fn$
declare n int := 0; c text;
begin
  for c in select distinct code from charters where not archived and coalesce(code,'')<>'' loop n := n + chat_sync(c); end loop;
  return n;
end $fn$;
grant execute on function public.chat_sync_all() to authenticated;

-- إضافة مباشرة (للمدير) — أو طلب انضمام (لغيره، ومنهم مسؤول الوحدة)
create or replace function public._chat_add(p_pcode text, p_user uuid) returns void
language plpgsql security definer set search_path=public as $fn$
declare ch bigint := _chat_ch(p_pcode);
begin
  delete from chat_excl where pcode=p_pcode and user_id=p_user;
  insert into chat_members(channel_id,user_id) values(ch,p_user) on conflict do nothing;
  update chat_join_requests set status='approved', decided_by=auth.uid(), decided_at=now() where pcode=p_pcode and user_id=p_user and status='pending';
  insert into user_nudges(to_user,from_user,from_name,msg)
    select p_user,auth.uid(),(select full_name from profiles where id=auth.uid()),'أُضفت إلى محادثة مشروع: '||coalesce((select name from chat_channels where id=ch),p_pcode) where p_user<>auth.uid();
end $fn$;
revoke all on function public._chat_add(text,uuid) from public;

create or replace function public.chat_join_request(p_pcode text, p_user uuid, p_why text default null) returns text
language plpgsql security definer set search_path=public as $fn$
declare adm boolean := exists(select 1 from profiles where id=auth.uid() and role='admin'); pn text; un text;
begin
  if not exists(select 1 from profiles where id=p_user) then raise exception 'bad user'; end if;
  if adm or chat_is_pm(p_pcode) then perform _chat_add(p_pcode,p_user); return 'added'; end if;
  if not (unit_head_of(p_pcode) or auth.uid() in (select public.chat_project_members(p_pcode))
          or exists(select 1 from chat_members m join chat_channels c on c.id=m.channel_id where c.kind='project' and c.pcode=p_pcode and m.user_id=auth.uid())) then
    raise exception 'not allowed'; end if;
  if exists(select 1 from chat_join_requests where pcode=p_pcode and user_id=p_user and status='pending') then return 'pending'; end if;
  insert into chat_join_requests(pcode,user_id,why,source,requested_by) values (p_pcode,p_user,nullif(trim(coalesce(p_why,'')),''),'request',auth.uid());
  pn := coalesce((select data->>'pname' from charters where code=p_pcode order by id desc limit 1), p_pcode);
  select full_name into un from profiles where id=p_user;
  perform _chat_notify_mgrs(p_pcode, 'طلب إضافة «'||coalesce(un,'')||'» إلى محادثة مشروع «'||pn||'». راجع طلبات الانضمام في التواصل.');
  return 'requested';
end $fn$;
grant execute on function public.chat_join_request(text,uuid,text) to authenticated;

create or replace function public.chat_join_decide(p_id bigint, p_ok boolean, p_note text default null) returns void
language plpgsql security definer set search_path=public as $fn$
declare q chat_join_requests%rowtype;
begin
  select * into q from chat_join_requests where id=p_id;
  if q.id is null or q.status<>'pending' then raise exception 'not found'; end if;
  if not (exists(select 1 from profiles where id=auth.uid() and role='admin') or chat_is_pm(q.pcode)) then raise exception 'not allowed'; end if;
  if p_ok then perform _chat_add(q.pcode, q.user_id); update chat_join_requests set note=p_note where id=p_id;
  else update chat_join_requests set status='rejected', decided_by=auth.uid(), decided_at=now(), note=p_note where id=p_id; end if;
end $fn$;
grant execute on function public.chat_join_decide(bigint,boolean,text) to authenticated;

create or replace function public.chat_join_pending(p_pcode text default null) returns table(id bigint, pcode text, pname text, user_id uuid, why text, source text, requested_by uuid, created_at timestamptz)
language sql stable security definer set search_path=public as $$
  select q.id, q.pcode, coalesce((select data->>'pname' from charters where code=q.pcode order by id desc limit 1), q.pcode), q.user_id, q.why, q.source, q.requested_by, q.created_at
  from chat_join_requests q
  where q.status='pending' and (p_pcode is null or q.pcode=p_pcode)
    and (exists(select 1 from profiles where id=auth.uid() and role='admin') or chat_is_pm(q.pcode))
  order by q.created_at desc
$$;
grant execute on function public.chat_join_pending(text) to authenticated;

-- الإضافة من داخل المحادثة: المدير يضيف مباشرة، وغيره يُنشئ طلبًا
drop function if exists public.chat_add_member(bigint,uuid);
create or replace function public.chat_add_member(p_cid bigint, p_user uuid) returns text
language plpgsql security definer set search_path=public as $fn$
declare pc text;
begin
  select pcode into pc from chat_channels where id=p_cid and kind='project';
  if pc is null or not chat_can_access(p_cid) then raise exception 'not allowed'; end if;
  return chat_join_request(pc, p_user, null);
end $fn$;
grant execute on function public.chat_add_member(bigint,uuid) to authenticated;

-- ===== (٣) المحادثات التي يراها الأدمن بصفته أدمن فقط (ليس عضوًا) =====
create or replace function public.chat_admin_only() returns table(channel_id bigint, pcode text, kind text)
language sql stable security definer set search_path=public as $$
  select c.id, c.pcode, c.kind from chat_channels c
  where exists(select 1 from profiles where id=auth.uid() and role='admin') and c.kind in ('project','group')
    and not exists(select 1 from chat_members m where m.channel_id=c.id and m.user_id=auth.uid())
    and (c.kind='group' or auth.uid() not in (select public.chat_project_members(c.pcode)))
  union all
  select null::bigint, x.code, 'project' from (select distinct code from charters where not archived and coalesce(code,'')<>'') x
  where exists(select 1 from profiles where id=auth.uid() and role='admin')
    and not exists(select 1 from chat_channels c where c.kind='project' and c.pcode=x.code)
    and auth.uid() not in (select public.chat_project_members(x.code))
$$;
grant execute on function public.chat_admin_only() to authenticated;

-- ===== (٤) رصد الأطراف الجدد تلقائيًا كل ساعة (إن كانت pg_cron متاحة) =====
do $$ begin
  create extension if not exists pg_cron;
  perform cron.unschedule(jobid) from cron.job where jobname='ido_chat_sync';
  perform cron.schedule('ido_chat_sync', '15 * * * *', 'select public.chat_sync_all()');
exception when others then raise notice 'pg_cron غير متاحة (%): يُرصد الأطراف الجدد عند فتح المحادثة وصفحة التحكم', sqlerrm; end $$;

-- ===== (١) فحص الصلاحية =====
create or replace function public.has_cap(k text) returns boolean
language sql stable security definer set search_path=public as $$
  select case
    when auth.uid() is null then false
    when exists(select 1 from profiles where id = auth.uid() and role = 'admin') then true
    when exists(select 1 from user_caps where user_id = auth.uid() and cap = k and mode = 'deny' and (expires_at is null or expires_at > now())) then false
    else coalesce(
      (select val = 'true' from user_caps where user_id = auth.uid() and cap = k and mode = 'set' and (expires_at is null or expires_at > now())),
      exists(select 1 from profiles p join app_roles a on a.key = coalesce(p.role_key,p.role) where p.id = auth.uid() and a.caps->>k = 'true')
      or exists(select 1 from user_roles x join app_roles a on a.key = x.role_key where x.user_id = auth.uid() and (x.expires_at is null or x.expires_at > now()) and a.caps->>k = 'true')
      or exists(select 1 from user_group_members m join user_groups g on g.id = m.group_id where m.user_id = auth.uid() and g.caps->>k = 'true')) end
$$;
grant execute on function public.has_cap(text) to authenticated;

create or replace function public.is_full_admin() returns boolean
language sql stable security definer set search_path=public as $$
  select exists(select 1 from profiles where id = auth.uid() and role = 'admin') $$;
grant execute on function public.is_full_admin() to authenticated;

-- إدارة الأدوار والصلاحيات
create or replace function public.perm_is_admin() returns boolean
language sql stable security definer set search_path=public as $$ select has_cap('adm.roles') $$;

-- صلاحية جديدة في الكتالوج: إعدادات المشاريع
update app_roles set caps = caps || '{"adm.projects":true}'::jsonb where base = 'admin' and caps is not null and not caps ? 'adm.projects';

-- ===== (٢) حماية: من يدير الصلاحيات جزئيًا لا يرفع نفسه ولا يمنح الأدمن الكامل =====
create or replace function public.perm_guard(p_uid uuid, p_role text, p_caps jsonb) returns void
language plpgsql stable security definer set search_path=public as $fn$
begin
  if is_full_admin() then return; end if;
  if p_uid = auth.uid() then raise exception 'cannot edit your own permissions'; end if;
  if exists(select 1 from profiles where id = p_uid and role = 'admin') then raise exception 'only a full admin can change a full admin'; end if;
  if p_role is not null and exists(select 1 from app_roles where key = p_role and base = 'admin') then raise exception 'only a full admin can grant the admin role'; end if;
  if p_caps is not null and exists(select 1 from jsonb_array_elements(p_caps) x where x->>'role' is not null and exists(select 1 from app_roles where key = x->>'role' and base = 'admin')) then
    raise exception 'only a full admin can grant the admin role'; end if;
end $fn$;

-- ملف الوصول والمعاينة: لمن يدير المستخدمين أو الصلاحيات
create or replace function public.perm_view_ok() returns boolean
language sql stable security definer set search_path=public as $$ select has_cap('adm.users') or has_cap('adm.roles') $$;
grant execute on function public.perm_view_ok() to authenticated;

do $g$
declare d text; f text;
begin
  -- نضيف الحماية في بداية دوال الصلاحيات (مرة واحدة)
  foreach f in array array['admin_user_access','perm_explain','perm_user_set','perm_bulk','perm_user_scope','perm_role_save','perm_role_delete','perm_revert','perm_group_caps'] loop
    select pg_get_functiondef(p.oid) into d from pg_proc p join pg_namespace n on n.oid = p.pronamespace where n.nspname = 'public' and p.proname = f limit 1;
    if d is null or d like '%perm_guard%' or d like '%is_full_admin()%' or d like '%perm_view_ok()%' then continue; end if;
    if f = 'perm_user_set' then
      d := regexp_replace(d, '(if not perm_is_admin\(\) then raise exception ''admin only''; end if;)', E'\\1\n  perform perm_guard(p_uid, p_role, p_extra);');
    elsif f = 'perm_user_scope' then
      d := regexp_replace(d, '(if not perm_is_admin\(\) then raise exception ''admin only''; end if;)', E'\\1\n  perform perm_guard(p_uid, null, null);');
    elsif f = 'perm_bulk' then
      d := regexp_replace(d, '(foreach u in array coalesce\(p_uids,''\{\}''\) loop)', E'\\1\n    perform perm_guard(u, p_role, null);');
    elsif f = 'perm_role_save' then
      d := regexp_replace(d, '(if not perm_is_admin\(\) then raise exception ''admin only''; end if;)', E'\\1\n  if not is_full_admin() and (p_base = ''admin'' or exists(select 1 from app_roles where key = p_key and base = ''admin'')) then raise exception ''only a full admin can edit admin roles''; end if;');
    elsif f = 'perm_role_delete' then
      d := regexp_replace(d, '(if not perm_is_admin\(\) then raise exception ''admin only''; end if;)', E'\\1\n  if not is_full_admin() and exists(select 1 from app_roles where key = p_key and base = ''admin'') then raise exception ''only a full admin can edit admin roles''; end if;');
    elsif f = 'perm_revert' then
      d := regexp_replace(d, '(if e\.id is null then raise exception ''not found''; end if;)', E'\\1\n  if not is_full_admin() and e.kind in (''user'',''scope'') then perform perm_guard(e.target::uuid, e.before->>''role_key'', null); end if;\n  if not is_full_admin() and e.kind in (''role'',''role_del'') and (e.before->>''base'' = ''admin'' or e.after->>''base'' = ''admin'') then raise exception ''only a full admin can edit admin roles''; end if;');
    elsif f in ('admin_user_access','perm_explain') then
      d := replace(d, 'if not perm_is_admin() then', 'if not perm_view_ok() then');
    elsif f = 'perm_group_caps' then
      d := regexp_replace(d, '(if not perm_is_admin\(\) then raise exception ''admin only''; end if;)', E'\\1\n  if not is_full_admin() and coalesce(p_caps,''{}'') ?| array(select key from (select jsonb_object_keys(p_caps) key) z where key like ''adm.%'') then raise exception ''only a full admin can grant admin permissions to groups''; end if;');
    end if;
    execute d;
  end loop;
end $g$;

-- خريطة الجداول (السياسات) ← الصلاحية
create or replace function public.perm_pol_cap(t text, cmd text, q text) returns table(expr text, h boolean)
language sql immutable as $$
  select * from (values (case
    when t in ('charters','plans','closure_reports','change_requests') and cmd = 'DELETE' then 'has_cap(''off.delete'')'
    when t in ('change_requests','signatories','audit_log','cr_attachments','attachments') then 'has_cap(''off.review'')'
    when t = 'del_requests' then 'has_cap(''off.delete'')'
    when t = 'escalations' then 'has_cap(''off.approve'')'
    when t in ('eval_model','project_evals','project_meta','portfolio_snapshots') then 'has_cap(''off.weights'')'
    when t in ('office_actions','ctl_check_history') then 'has_cap(''adm.control'')'
    when t in ('escalation_rules','escalation_log','approver_leave') then 'has_cap(''off.escalate'')'
    when t in ('project_cfg','project_links','project_handover','project_transfer') then 'has_cap(''adm.projects'')'
    when t in ('project_grants','project_blocks','profiles') then 'has_cap(''adm.users'')'
    when t in ('user_groups','user_group_members','app_roles') then 'has_cap(''adm.roles'')'
    when t in ('app_settings','app_config') then 'has_cap(''adm.settings'')'
    when t = 'objects' and q like '%templates%' then 'has_cap(''adm.settings'')'
    when t like 'chat\_%' then 'has_cap(''chat.all'')'
    when t in ('service_ratings','ratings','ideas','idea_votes') then 'has_cap(''adm.control'')'
    when t like '%audit%' then 'has_cap(''audit.view'')'
    end,
    (t in ('charters','plans','closure_reports','change_requests') and cmd = 'DELETE')
      or t in ('eval_model','project_evals','project_meta','office_actions','ctl_check_history'))) v(expr, h)
$$;

-- ===== (٥) التطبيق =====
-- استبدال دقيق لفحص «exists(select 1 from profiles where id = auth.uid() and role = 'admin')» مع موازنة الأقواس
create or replace function public.perm_sub(src text, expr text) returns text
language plpgsql immutable as $fn$
declare
  rx text := '\mexists\s*\(\s*select\s+1\s+from\s+(public\.)?profiles(\s+\w+)?\s+where\s+\(*\s*(\w+\.)?id\s*=\s*auth\.uid\(\)\s*\)*\s+and\s+\(*\s*(\w+\.)?role\s*=\s*''admin''(::text)?';
  o text := ''; rest text := src; i int; m text; j int; dep int; k int; ch text; tail text; guard int := 0;
begin
  if src is null then return null; end if;
  loop
    guard := guard + 1; exit when guard > 200;
    i := regexp_instr(rest, rx, 1, 1, 0, 'i');
    exit when i = 0;
    m := substring(rest from i for (regexp_instr(rest, rx, 1, 1, 1, 'i') - i));
    j := i + position('(' in substring(rest from i)) - 1;          -- قوس exists
    dep := 0; k := j;
    loop
      ch := substr(rest, k, 1);
      exit when ch = '';
      if ch = '(' then dep := dep + 1; elsif ch = ')' then dep := dep - 1; end if;
      exit when dep = 0;
      k := k + 1;
    end loop;
    tail := substr(rest, i + length(m), k - (i + length(m)));
    if ch = ')' and dep = 0 and tail ~ '^[\s)]*$' then
      o := o || substr(rest, 1, i - 1) || expr; rest := substr(rest, k + 1);
    else
      o := o || substr(rest, 1, i + length(m) - 1); rest := substr(rest, i + length(m));
    end if;
  end loop;
  return o || rest;
end $fn$;
create or replace function public.perm_rx_helper() returns text language sql immutable as $$
  select '(public\.)?\m(pl_is_admin|ch_is_admin|cl_is_admin|_is_admin)\(\)' $$;

create or replace function public.perm_phase2_apply() returns jsonb
language plpgsql security definer set search_path=public as $fn$
declare r record; d text; n text; m record; q text; w text; nq text; nw text; nf int := 0; np int := 0; c record; tq text;
begin
  if auth.uid() is not null and not is_full_admin() then raise exception 'admin only'; end if;
  -- الدوال
  for r in select p.oid, p.proname, p.oid::regprocedure::text sig, fm.expr, fm.h from pg_proc p join pg_namespace ns on ns.oid = p.pronamespace
           join perm_fn_map fm on fm.fn = p.proname where ns.nspname = 'public' loop
    select def into d from perm_fn_backup where sig = r.sig;
    if d is null then d := pg_get_functiondef(r.oid); insert into perm_fn_backup(sig, cap, def) values(r.sig, r.expr, d); end if;
    n := perm_sub(d, r.expr);
    if r.h then n := regexp_replace(n, perm_rx_helper(), r.expr, 'g'); end if;
    if n <> pg_get_functiondef(r.oid) then
      begin execute n; nf := nf + 1; insert into perm_phase2_report(kind,name,cap,status) values('function', r.proname, r.expr, 'updated');
      exception when others then insert into perm_phase2_report(kind,name,cap,status) values('function', r.proname, r.expr, 'error: '||sqlerrm); end;
    elsif n = d then insert into perm_phase2_report(kind,name,cap,status) values('function', r.proname, r.expr, 'no admin check found');
    end if;
  end loop;
  -- السياسات
  for c in select schemaname, tablename, policyname, cmd, qual, with_check from pg_policies where schemaname in ('public','storage') loop
    tq := coalesce(c.qual,'') || ' ' || coalesce(c.with_check,'');
    if tq !~* 'admin' then continue; end if;
    select * into m from perm_pol_cap(c.tablename, c.cmd, tq);
    if m.expr is null then continue; end if;
    select qual, wchk into q, w from perm_pol_backup where tbl = c.schemaname||'.'||c.tablename and pol = c.policyname;
    if not found then q := c.qual; w := c.with_check;
      insert into perm_pol_backup(tbl,pol,cmd,qual,wchk,cap) values(c.schemaname||'.'||c.tablename, c.policyname, c.cmd, q, w, m.expr); end if;
    nq := q; nw := w;
    if nq is not null then nq := perm_sub(nq, m.expr); if m.h then nq := regexp_replace(nq, perm_rx_helper(), m.expr, 'g'); end if; end if;
    if nw is not null then nw := perm_sub(nw, m.expr); if m.h then nw := regexp_replace(nw, perm_rx_helper(), m.expr, 'g'); end if; end if;
    if nq is distinct from c.qual or nw is distinct from c.with_check then
      begin
        execute format('alter policy %I on %I.%I', c.policyname, c.schemaname, c.tablename)
             || case when nq is not null then ' using (' || nq || ')' else '' end
             || case when nw is not null then ' with check (' || nw || ')' else '' end;
        np := np + 1; insert into perm_phase2_report(kind,name,cap,status) values('policy', c.tablename||' / '||c.policyname, m.expr, 'updated');
      exception when others then insert into perm_phase2_report(kind,name,cap,status) values('policy', c.tablename||' / '||c.policyname, m.expr, 'error: '||sqlerrm); end;
    end if;
  end loop;
  return jsonb_build_object('functions', nf, 'policies', np);
end $fn$;

create or replace function public.perm_phase2_rollback() returns jsonb
language plpgsql security definer set search_path=public as $fn$
declare r record; nf int := 0; np int := 0; s text; t text;
begin
  if auth.uid() is not null and not is_full_admin() then raise exception 'admin only'; end if;
  for r in select * from perm_fn_backup loop
    begin execute r.def; nf := nf + 1; exception when others then raise notice 'skip %: %', r.sig, sqlerrm; end;
  end loop;
  for r in select * from perm_pol_backup loop
    s := split_part(r.tbl,'.',1); t := split_part(r.tbl,'.',2);
    begin
      execute format('alter policy %I on %I.%I', r.pol, s, t)
           || case when r.qual is not null then ' using (' || r.qual || ')' else '' end
           || case when r.wchk is not null then ' with check (' || r.wchk || ')' else '' end;
      np := np + 1;
    exception when others then raise notice 'skip %: %', r.pol, sqlerrm; end;
  end loop;
  delete from perm_fn_backup; delete from perm_pol_backup;
  return jsonb_build_object('functions_restored', nf, 'policies_restored', np);
end $fn$;
revoke all on function public.perm_phase2_apply(), public.perm_phase2_rollback() from public, anon;
grant execute on function public.perm_phase2_apply(), public.perm_phase2_rollback() to authenticated;

truncate public.perm_phase2_report;
select perm_phase2_apply() as result;

notify pgrst, 'reload schema';

select kind as "النوع", name as "الاسم", cap as "الصلاحية", status as "الحالة" from perm_phase2_report order by kind, status desc, name;

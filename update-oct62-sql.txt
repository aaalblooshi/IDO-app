-- =====================================================================
-- update-oct62: حماية الدور في ملف المستخدم (يُشغَّل بعد oct61)
--   من يملك «إدارة المستخدمين» يستطيع تعديل ملفات المستخدمين، فنمنع تغيير الدور عبر التعديل المباشر:
--   • الأدمن الكامل: بلا قيود
--   • من يملك «الأدوار والصلاحيات»: يغيّر دور غيره فقط، وليس إلى أدمن ولا من أدمن
--   • غير ذلك: لا يتغير الدور (ولا يستطيع أي مستخدم رفع دوره بنفسه)
-- =====================================================================
create or replace function public.profiles_role_guard() returns trigger
language plpgsql security definer set search_path=public as $fn$
begin
  if new.role is not distinct from old.role and new.role_key is not distinct from old.role_key then return new; end if;
  if auth.uid() is null or is_full_admin() then return new; end if;
  if has_cap('adm.roles') and new.id <> auth.uid()
     and coalesce(old.role,'') <> 'admin' and coalesce(new.role,'') <> 'admin'
     and not exists(select 1 from app_roles where key = new.role_key and base = 'admin') then
    return new;
  end if;
  raise exception 'only a full admin can change this role';
end $fn$;

drop trigger if exists profiles_role_guard on public.profiles;
create trigger profiles_role_guard before update on public.profiles
  for each row execute function public.profiles_role_guard();

notify pgrst, 'reload schema';

create or replace function public.chat_involved(p_pcode text) returns table(user_id uuid, why text)
language sql stable security definer set search_path=public as $$
  select distinct on (u) u, w from (
    select pl.created_by u, 'أعدّ خطة المشروع' w from plans pl where pl.data->>'pcode'=p_pcode and not pl.archived
    union all select s.user_id, 'معدّ الميثاق' from charter_signatories s join charters c on c.id=s.cr_id where c.code=p_pcode and s.stage not in ('reviewed','approved')
    union all select s.user_id, 'معدّ الخطة' from plan_signatories s join plans pl on pl.id=s.cr_id where pl.data->>'pcode'=p_pcode and s.stage not in ('reviewed','approved')
    union all select case when t->>'w' ~ '^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$' then (t->>'w')::uuid end, 'مكلّف بمهمة: '||left(coalesce(t->>'n',''),60)
              from plans pl, jsonb_array_elements(coalesce(pl.data->'tasks','[]'::jsonb)) t where pl.data->>'pcode'=p_pcode and not pl.archived
    union all select g.user_id, 'مُنح صلاحية على المشروع' from project_grants g where g.pcode=p_pcode
    union all select unnest(k.invitees), 'مشارك في مراجعة الخطة' from plan_consult k join plans pl on pl.id=k.plan_id where pl.data->>'pcode'=p_pcode
  ) z where u is not null order by u
$$;
revoke all on function public.chat_involved(text) from public;

-- إلغاء الطلبات المعلّقة التي نشأت فقط لأن الشخص مراجع أو معتمد
delete from public.chat_join_requests q
where q.status='pending' and q.source='auto' and q.why like 'موقّع%'
  and not exists(select 1 from public.chat_involved(q.pcode) i where i.user_id=q.user_id);
update public.chat_join_requests q set why = i.why
from (select r.id, (select i.why from public.chat_involved(r.pcode) i where i.user_id=r.user_id) why from public.chat_join_requests r where r.status='pending' and r.source='auto' and r.why like 'موقّع%') i
where q.id = i.id and i.why is not null;
notify pgrst, 'reload schema';

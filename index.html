create extension if not exists pg_net; create table if not exists app_config(key text primary key, value text); alter table app_config enable row level security; insert into app_config values ('mail_url','PASTE_WEB_APP_URL'), ('mail_secret','SAME_SECRET'), ('site_url','https://aaalblooshi.github.io/cr-app/') on conflict (key) do update set value=excluded.value; create or replace function notify_signatory() returns trigger language plpgsql security definer set search_path=public as $fn$ declare u profiles; c change_requests; url text; sec text; site text; begin select * into u from profiles where id=new.user_id; select * into c from change_requests where id=new.cr_id; select value into url from app_config where key='mail_url'; select value into sec from app_config where key='mail_secret'; select value into site from app_config where key='site_url'; if url is null or u.email is null then return new; end if; perform net.http_post(url:=url, body:=jsonb_build_object('secret',sec,'to',u.email,'name',u.full_name,'stage',new.stage,'code',c.code,'project',c.data->>'pname','site',site), headers:='{"Content-Type":"application/json"}'::jsonb); return new; exception when others then return new; end $fn$; drop trigger if exists trg_notify on signatories; create trigger trg_notify after insert on signatories for each row execute function notify_signatory();
